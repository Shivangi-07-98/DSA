https://leetcode.com/problems/find-minimum-in-rotated-sorted-array/

time O(log n)
space O(1)

class Solution {
  public int findMin(int[] nums) {
    int lo = 0;
    int hi = nums.length - 1;

    while (lo < hi) {
      int mid = (lo + hi) / 2;

      if (nums[hi] > nums[mid]) {
        hi = mid;
      } else {
        lo = mid + 1;
      }
    }

    return nums[lo];
  }
}
